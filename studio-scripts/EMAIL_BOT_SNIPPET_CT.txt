// ═══════════════════════════════════════════════════════════════
// CT EMAIL BOT — Studio Snippet (with heartbeat)
// ═══════════════════════════════════════════════════════════════
// Fires a fast heartbeat GET FIRST to guarantee the email is logged
// even if the main POST later times out. Then calls the bot normally.
// ═══════════════════════════════════════════════════════════════

// ── Block known spam senders ──
ASSIGN blockEmail = "false"
ASSIGN emailFromLower = "{emailFrom.lower()}"
ASSIGN isSpam = emailFromLower.contains('autos1000.com')
IF isSpam = 1
{
   ASSIGN blockEmail = "true"
}

IF blockEmail = "true"
{
   ASSIGN botSpam = "true"
   ASSIGN botReply = ""
   ASSIGN botEscalate = "false"
}
ELSE
{
   // Default values — used if main bot call fails
   ASSIGN botReply = ""
   ASSIGN botEscalate = "true"
   ASSIGN botSpam = "false"

   // ── Step 1: Fire heartbeat (fast, fire-and-forget) ──
   ASSIGN hbProxy = GetRESTProxy()
   ASSIGN hbProxy.ContentType = "application/json"
   ASSIGN hbProxy.ProxyTimeoutSeconds = 10
   ASSIGN hbUrl = "https://cxone-faq-bot-1.onrender.com/api/heartbeat?contactId={ContactId}&from={emailFrom}&state=CT&subject={emailSubject}"
   DYNAMIC hbResult
   ASSIGN hbResult = hbProxy.MakeRestRequest(hbUrl, "", 0, "GET")

   // ── Step 2: Main bot call ──
   ASSIGN proxy = GetRESTProxy()
   ASSIGN proxy.ContentType = "application/json"
   ASSIGN proxy.ProxyTimeoutSeconds = 90
   ASSIGN botUrl = "https://cxone-faq-bot-1.onrender.com/api/email?api_key=tns-bot-secret-2024"

   DYNAMIC requestBody
   ASSIGN requestBody.contactId = ContactId
   ASSIGN requestBody.from = emailFrom
   ASSIGN requestBody.subject = emailSubject
   ASSIGN requestBody.body = emailBody
   ASSIGN requestBody.state = "CT"
   ASSIGN payload = "{requestBody.asJSON()}"

   DYNAMIC result
   ASSIGN result = proxy.MakeRestRequest(botUrl, payload, 0, "POST")

   // If we got a response, use it. Otherwise keep defaults (escalate).
   IF result.confidence != ""
   {
      ASSIGN botReply = "{result.reply}"
      ASSIGN botEscalate = "{result.escalate}"
      ASSIGN botSpam = "{result.spam}"
   }
}
