// ═══════════════════════════════════════════════════════════════
// MA EMAIL BOT — Studio Snippet
// ═══════════════════════════════════════════════════════════════
// Includes:
// - Autos1000 spam blocking
// - Direct variable passing (no {var} substitution) to prevent JSON errors
// - Fallback failure logging if the bot is down / times out
// ═══════════════════════════════════════════════════════════════

// ── Block unwanted email senders ──
ASSIGN blockEmail = "false"
ASSIGN emailFromLower = "{emailFrom.lower()}"
ASSIGN isSpam = emailFromLower.contains('autos1000.com')
IF isSpam = 1
{
   ASSIGN blockEmail = "true"
}

IF blockEmail = "true"
{
   ASSIGN botSpam = "true"
   ASSIGN botReply = ""
   ASSIGN botEscalate = "false"
}
ELSE
{
   // Default values — will be used if the bot call fails
   ASSIGN botReply = ""
   ASSIGN botEscalate = "true"
   ASSIGN botSpam = "false"

   // Call the bot
   ASSIGN proxy = GetRESTProxy()
   ASSIGN proxy.ContentType = "application/json"
   ASSIGN proxy.ProxyTimeoutSeconds = 90
   ASSIGN botUrl = "https://cxone-faq-bot-1.onrender.com/api/email?api_key=tns-bot-secret-2024"

   DYNAMIC requestBody
   ASSIGN requestBody.contactId = ContactId
   ASSIGN requestBody.from = emailFrom
   ASSIGN requestBody.subject = emailSubject
   ASSIGN requestBody.body = emailBody
   ASSIGN requestBody.state = "MA"
   ASSIGN payload = "{requestBody.asJSON()}"

   DYNAMIC result
   ASSIGN result = proxy.MakeRestRequest(botUrl, payload, 0, "POST")

   ASSIGN gotResult = "false"
   IF result.confidence != ""
   {
      ASSIGN gotResult = "true"
   }

   IF gotResult = "true"
   {
      ASSIGN botReply = "{result.reply}"
      ASSIGN botEscalate = "{result.escalate}"
      ASSIGN botSpam = "{result.spam}"
   }
   ELSE
   {
      // Bot call failed — log the miss so it shows up on dashboard
      ASSIGN failUrl = "https://cxone-faq-bot-1.onrender.com/api/email-failed?contactId={ContactId}&from={emailFrom}&state=MA&subject={emailSubject}"
      DYNAMIC failResult
      ASSIGN failResult = proxy.MakeRestRequest(failUrl, "", 0, "GET")
   }
}
